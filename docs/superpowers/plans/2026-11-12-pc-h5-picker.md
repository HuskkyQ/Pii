/** 桌面浮层定位常量，与 spec 第 4.1 节一致。 */
export const PICKER_WIDTH = 360;
export const PICKER_GAP = 8;
export const PICKER_EDGE = 12;

export type PickerAlign = "start" | "end";

export function desiredPickerLeft(
  rect: { left: number; right: number },
  align: PickerAlign,
  width: number = PICKER_WIDTH,
): number {
  return align === "end" ? rect.right - width : rect.left;
}

export function clampPickerLeft(
  desired: number,
  viewportWidth: number,
  width: number = PICKER_WIDTH,
  edge: number = PICKER_EDGE,
): number {
  const maxLeft = Math.max(edge, viewportWidth - width - edge);
  return Math.max(edge, Math.min(desired, maxLeft));
}

export function resolvePickerLeft(
  rect: { left: number; right: number },
  align: PickerAlign,
  viewportWidth: number,
  width: number = PICKER_WIDTH,
  edge: number = PICKER_EDGE,
): number {
  return clampPickerLeft(desiredPickerLeft(rect, align, width), viewportWidth, width, edge);
}

export function pickerBottom(
  anchorTop: number,
  viewportHeight: number,
  gap: number = PICKER_GAP,
): number {
  return viewportHeight - anchorTop + gap;
}
```

import { useCallback, useSyncExternalStore } from "react";

export function useMediaQuery(query: string): boolean {
  const subscribe = useCallback(
    (onChange: () => void) => {
      const mql = window.matchMedia(query);
      mql.addEventListener("change", onChange);
      return () => mql.removeEventListener("change", onChange);
    },
    [query],
  );

  const getSnapshot = useCallback(() => window.matchMedia(query).matches, [query]);

  return useSyncExternalStore(subscribe, getSnapshot, () => false);
}
```

- [ ] **Step 2: 创建 `responsive-picker.tsx`**

```tsx
import { useEffect, useMemo, useRef, useState, type ReactNode } from "react";

import { useMediaQuery } from "@/hooks/useMediaQuery";
import { pickerBottom, resolvePickerLeft, type PickerAlign } from "@/lib/picker-position";

const DESKTOP_QUERY = "(min-width: 768px)";

interface ResponsivePickerProps {
  open: boolean;
  anchor?: HTMLElement | null;
  align?: PickerAlign;
  label: string;
  mobileMaxHeight?: string;
  onClose: () => void;
  children: (ctx: { isDesktop: boolean }) => ReactNode;
}

interface Viewport {
  width: number;
  height: number;
}

interface PanelLayout {
  left: number;
  bottom: number;
}

/**
 * 三个选择器共用的双端外壳。
 * 桌面：贴 anchor 的 360px 浮层 + 透明点击层；手机：原样底部抽屉。
 */
export function ResponsivePicker({
  open,
  anchor,
  align = "start",
  label,
  mobileMaxHeight = "82dvh",
  onClose,
  children,
}: ResponsivePickerProps) {
  const isDesktop = useMediaQuery(DESKTOP_QUERY);
  const panelRef = useRef<HTMLDivElement>(null);
  const wasOpen = useRef(false);
  const [viewport, setViewport] = useState<Viewport>(() => ({
    width: window.innerWidth,
    height: window.innerHeight,
  }));

  useEffect(() => {
    const onResize = () => {
      setViewport({ width: window.innerWidth, height: window.innerHeight });
    };
    window.addEventListener("resize", onResize);
    window.visualViewport?.addEventListener("resize", onResize);
    return () => {
      window.removeEventListener("resize", onResize);
      window.visualViewport?.removeEventListener("resize", onResize);
    };
  }, []);

  useEffect(() => {
    if (!open) return;
    const onKeyDown = (event: KeyboardEvent) => {
      if (event.key === "Escape") onClose();
    };
    window.addEventListener("keydown", onKeyDown);
    return () => window.removeEventListener("keydown", onKeyDown);
  }, [open, onClose]);

  useEffect(() => {
    if (!open) return;
    const frame = requestAnimationFrame(() => {
      const target = panelRef.current?.querySelector<HTMLElement>(
        'input, button, [tabindex]:not([tabindex="-1"])',
      );
      target?.focus();
    });
    return () => cancelAnimationFrame(frame);
  }, [open, isDesktop]);

  useEffect(() => {
    if (wasOpen.current && !open) anchor?.focus();
    wasOpen.current = open;
  }, [open, anchor]);

  const layout = useMemo<PanelLayout | null>(() => {
    if (!open || !isDesktop) return null;
    const rect = anchor?.getBoundingClientRect();
    const valid =
      rect !== undefined &&
      (rect.width > 0 || rect.height > 0 || rect.left !== 0 || rect.top !== 0);
    if (!rect || !valid) return null;
    return {
      left: resolvePickerLeft(rect, align, viewport.width),
      bottom: pickerBottom(rect.top, viewport.height),
    };
  }, [open, isDesktop, anchor, align, viewport.width, viewport.height]);

  const content = children({ isDesktop });

  if (!open) return null;

  const inner = <div className="flex min-h-0 flex-1 flex-col">{content}</div>;
  const panelClass =
    "flex max-h-[min(70dvh,520px)] w-[360px] max-w-[calc(100vw-24px)] flex-col rounded-xl border bg-background shadow-2xl";

  if (isDesktop) {
    return (
      <div className="fixed inset-0 z-[70]">
        <div className="absolute inset-0 z-[69] bg-transparent" onClick={onClose} aria-hidden />
        {layout ? (
          <div
            ref={panelRef}
            role="dialog"
            aria-modal="true"
            aria-label={label}
            style={{ left: layout.left, bottom: layout.bottom }}
            className={`fixed z-[70] ${panelClass}`}
          >
            {inner}
          </div>
        ) : (
          <div
            ref={panelRef}
            role="dialog"
            aria-modal="true"
            aria-label={label}
            className={`fixed left-1/2 top-1/2 z-[70] -translate-x-1/2 -translate-y-1/2 ${panelClass}`}
          >
            {inner}
          </div>
        )}
      </div>
    );
  }

  return (
    <div className="fixed inset-0 z-[70]">
      <div className="absolute inset-0 bg-black/50" onClick={onClose} aria-hidden />
      <div
        ref={panelRef}
        role="dialog"
        aria-modal="true"
        aria-label={label}
        style={{ maxHeight: mobileMaxHeight }}
        className="absolute inset-x-0 bottom-0 flex flex-col rounded-t-2xl border-t bg-background pb-[env(safe-area-inset-bottom)] shadow-2xl"
      >
        {inner}
      </div>
    </div>
  );
}
```

import { Button } from "@/components/ui/button";

import { KNOWN_TOOLS, type KnownTool, type PresetId, type ToolPolicy } from "@/lib/types";

interface PermissionPickerProps {
  open: boolean;
  anchor?: HTMLElement | null;
  /** 当前会话策略；null = 还在加载 */
  policy: ToolPolicy | null;
  preset: PresetId | null;
  onApply: (policy: ToolPolicy) => Promise<void>;

/**
 * 工具审批策略选择器。
 * 桌面：贴「权限」chip 的小浮层；手机：底部抽屉。
 * 桌面点预设保存后自动收起，高级逐工具开关保持打开；手机仍点「完成」。
 */
export function PermissionPicker({ open, anchor, policy, preset, onApply, onClose }: PermissionPickerProps) {
  const [advanced, setAdvanced] = useState(false);
  const [busy, setBusy] = useState(false);

  // 按当前 policy 推断预设（以本地 policy 为准，比服务端 preset 更实时）
  const currentPreset: PresetId = policy ? presetOf(policy) : (preset ?? "read-only");
  const danger = currentPreset === "all";

  async function apply(next: ToolPolicy): Promise<boolean> {
    if (busy) return false;
    setBusy(true);
    try {
      await onApply(next);
      return true;
    } catch (error) {
      toast.error(error instanceof Error ? error.message : "更新权限失败");
      return false;
    } finally {
      setBusy(false);
    }
  }

  async function applyPreset(id: Exclude<PresetId, "custom">, isDesktop: boolean) {
    const ok = await apply(PRESETS[id]);
    if (ok && isDesktop) onClose();
  }

  function toggleTool(tool: KnownTool) {
    if (!policy) return;
    // 手动逐工具时不再保留 allowUnknown（更安全：未知工具仍问）
    void apply({ auto: { ...policy.auto, [tool]: !policy.auto[tool] }, allowUnknown: false });
  }

  return (
    <ResponsivePicker
      open={open}
      anchor={anchor}
      align="start"
      label="工具执行权限"
      mobileMaxHeight="82dvh"
      onClose={onClose}
    >
      {({ isDesktop }) => (
        <>
          <div className="flex shrink-0 items-center gap-2 px-3 pb-2 pt-3">
            {danger ? (

import { Check, ImageOff, Loader2, Search, Sparkles, X } from "lucide-react";
import { useEffect, useMemo, useRef, useState } from "react";
import { toast } from "sonner";

import { Button } from "@/components/ui/button";
import { ResponsivePicker } from "@/components/ui/responsive-picker";
import { listModels, setModel } from "@/lib/api";
import type { ModelOption } from "@/lib/types";
import { cn } from "@/lib/utils";

interface ModelPickerProps {
  open: boolean;
  anchor?: HTMLElement | null;
  conversationId?: number;
  current?: { provider?: string; modelId?: string } | null;
  onClose: () => void;
  onPicked: () => void;
}

/** 模型选择器：桌面贴「模型」chip 右对齐浮层，手机仍是底部抽屉。 */
export function ModelPicker({ open, anchor, conversationId, current, onClose, onPicked }: ModelPickerProps) {
  const [models, setModels] = useState<ModelOption[]>([]);
  const [query, setQuery] = useState("");
  const [busyKey, setBusyKey] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    if (!open) return;
    setLoading(true);
    listModels(conversationId)
      .then((data) => setModels(data.models))
      .catch((error: unknown) => toast.error(error instanceof Error ? error.message : String(error)))
      .finally(() => setLoading(false));
    // 打开时聚焦搜索框：手机上少点一下
    setTimeout(() => inputRef.current?.focus(), 250);
  }, [open, conversationId]);

  const groups = useMemo(() => {
    const keyword = query.trim().toLowerCase();
    const filtered = keyword
      ? models.filter(
          (model) =>
            model.name.toLowerCase().includes(keyword) ||
            model.id.toLowerCase().includes(keyword) ||
            model.provider.toLowerCase().includes(keyword),
        )
      : models;

    const byProvider = new Map<string, ModelOption[]>();
    for (const model of filtered) {
      const list = byProvider.get(model.provider) ?? [];
      list.push(model);
      byProvider.set(model.provider, list);
    }
    return [...byProvider.entries()].sort(([a], [b]) => a.localeCompare(b));
  }, [models, query]);

  async function pick(model: ModelOption) {
    const key = `${model.provider}/${model.id}`;
    const isCurrent = current?.provider === model.provider && current?.modelId === model.id;
    if (isCurrent) return onClose();

    setBusyKey(key);
    try {
      await setModel(model.provider, model.id, conversationId);
      toast.success(`已切到 ${model.name}`);
      onPicked();
      onClose();
    } catch (error) {
      toast.error(error instanceof Error ? error.message : String(error));
    } finally {
      setBusyKey(null);
    }
  }

  return (
    <ResponsivePicker
      open={open}
      anchor={anchor}
      align="end"
      label="切换模型"
      mobileMaxHeight="78dvh"
      onClose={onClose}
    >
      {() => (
        <>
          <div className="flex shrink-0 items-center gap-2 px-3 pb-2 pt-3">
            <span className="text-[13px] font-medium">切换模型</span>
            <span className="text-[11px] text-muted-foreground">{models.length} 个可用</span>
            <Button size="icon" variant="ghost" className="ml-auto size-8" onClick={onClose} aria-label="关闭">
              <X className="size-4" />
            </Button>
          </div>

          <div className="mx-3 mb-2 flex shrink-0 items-center gap-2 rounded-lg border bg-muted/40 px-2.5">
            <Search className="size-3.5 shrink-0 text-muted-foreground" aria-hidden />
            <input
              ref={inputRef}
              value={query}
              onChange={(event) => setQuery(event.target.value)}
              placeholder="搜模型名 / provider"
              className="h-9 min-w-0 flex-1 bg-transparent text-[13px] outline-none placeholder:text-muted-foreground"
            />
          </div>

          <div className="min-h-0 flex-1 overflow-y-auto overscroll-contain px-2 pb-3">
            {loading && (
              <div className="flex items-center gap-2 px-2 py-6 text-[12.5px] text-muted-foreground">
                <Loader2 className="size-3.5 animate-spin" aria-hidden /> 读取模型列表…
              </div>
            )}

            {!loading &&
              groups.map(([provider, list]) => (
                <div key={provider} className="mt-2">
                  <div className="sticky top-0 z-10 bg-background/95 px-2 py-1 text-[11px] uppercase tracking-wider text-muted-foreground/70 backdrop-blur">
                    {provider}
                  </div>
                  {list.map((model) => {
                    const active = current?.provider === model.provider && current?.modelId === model.id;
                    const pending = busyKey === `${model.provider}/${model.id}`;
                    return (
                      <button
                        key={`${model.provider}/${model.id}`}
                        type="button"
                        onClick={() => void pick(model)}
                        className={cn(
                          "mt-0.5 flex w-full items-center gap-2 rounded-md px-2 py-2 text-left transition-colors",
                          active ? "bg-accent text-accent-foreground" : "hover:bg-accent/50",
                        )}
                      >
                        <span className="min-w-0 flex-1">
                          <span className="flex items-center gap-1.5">
                            <span className="truncate text-[13px]">{model.name}</span>
                            {model.reasoning && (
                              <Sparkles className="size-3 shrink-0 text-amber-400/80" aria-label="支持思考" />
                            )}
                            {!model.image && (
                              <ImageOff className="size-3 shrink-0 text-muted-foreground/50" aria-label="不支持图片" />
                            )}
                          </span>
                          <span className="block truncate font-mono text-[11px] text-muted-foreground">
                            {model.id}
                            {model.contextWindow ? ` · ${Math.round(model.contextWindow / 1000)}k` : ""}
                          </span>
                        </span>
                        {pending ? (
                          <Loader2 className="size-4 shrink-0 animate-spin" aria-hidden />
                        ) : (
                          active && <Check className="size-4 shrink-0" aria-hidden />
                        )}
                      </button>
                    );
                  })}
                </div>
              ))}

            {!loading && groups.length === 0 && (
              <p className="px-2 py-8 text-center text-[12.5px] text-muted-foreground">没有匹配的模型</p>
            )}
          </div>
        </>
      )}
    </ResponsivePicker>
  );
}

import { ChevronLeft, FolderOpen, Home, Loader2, X } from "lucide-react";
import { useEffect, useRef, useState } from "react";
import { toast } from "sonner";

import { Button } from "@/components/ui/button";
import { ResponsivePicker } from "@/components/ui/responsive-picker";
import { browseDir, getCwd, setCwd } from "@/lib/api";
import type { DirListing } from "@/lib/types";

interface CwdPickerProps {
  open: boolean;
  anchor?: HTMLElement | null;
  conversationId?: number;
  onClose: () => void;
  onPicked: (cwd: string) => void;
}

/** 选工作目录：桌面贴「📁 工作目录」浮层，手机仍是底部抽屉。 */
export function CwdPicker({ open, anchor, conversationId, onClose, onPicked }: CwdPickerProps) {
  const [listing, setListing] = useState<DirListing | null>(null);
  const [current, setCurrent] = useState<string>("");
  const [draft, setDraft] = useState("");
  const [busy, setBusy] = useState(false);
  const inputRef = useRef<HTMLInputElement>(null);

  const load = async (path?: string) => {
    setBusy(true);
    try {
      const next = await browseDir(path);
      setListing(next);
      setCurrent(next.path);
      setDraft(next.path);
    } catch (error) {
      toast.error(error instanceof Error ? error.message : String(error));
    } finally {
      setBusy(false);
    }
  };

  useEffect(() => {
    if (!open) return;
    getCwd(conversationId)
      .then((data) => load(data.cwd))
      .catch(() => load());
  }, [open, conversationId]);

  async function confirm(path: string) {
    setBusy(true);
    try {
      const result = await setCwd(path, conversationId);
      toast.success(`工作目录 → ${result.cwd.split("/").slice(-1)[0] || result.cwd}`);
      onPicked(result.cwd);
      onClose();
    } catch (error) {
      toast.error(error instanceof Error ? error.message : String(error));
    } finally {
      setBusy(false);
    }
  }

  return (
    <ResponsivePicker
      open={open}
      anchor={anchor}
      align="start"
      label="选择工作目录"
      mobileMaxHeight="78dvh"
      onClose={onClose}
    >
      {() => (
        <>
          <div className="flex shrink-0 items-center gap-2 px-3 pb-2 pt-3">
            <FolderOpen className="size-4 shrink-0 text-muted-foreground" aria-hidden />
            <span className="text-[13px] font-medium">选择工作目录</span>
            <span className="text-[11px] text-muted-foreground">agent 只在这个目录里读写</span>
            <Button size="icon" variant="ghost" className="ml-auto size-8" onClick={onClose} aria-label="关闭">
              <X className="size-4" />
            </Button>
          </div>

          {/* 路径输入：桌面端可以直接粘绝对路径 */}
          <div className="mx-3 mb-2 flex shrink-0 items-center gap-2 rounded-lg border bg-muted/40 px-2">
            <input
              ref={inputRef}
              value={draft}
              onChange={(event) => setDraft(event.target.value)}
              onKeyDown={(event) => {
                if (event.key === "Enter") {
                  event.preventDefault();
                  void load(draft);
                }
              }}
              spellCheck={false}
              placeholder="/Users/you/projects/foo"
              className="h-9 min-w-0 flex-1 bg-transparent font-mono text-[12px] outline-none placeholder:text-muted-foreground/60"
            />
            <Button size="sm" variant="ghost" className="h-7 px-2 text-[12px]" onClick={() => void load(draft)}>
              前往
            </Button>
          </div>

          <div className="min-h-0 flex-1 overflow-y-auto overscroll-contain px-2 pb-2">
            {listing && (
              <>
                <div className="flex items-center gap-1 px-1 py-1">
                  {listing.parent && (
                    <Button
                      size="sm"
                      variant="ghost"
                      className="h-7 gap-1 px-2 text-[12px]"
                      onClick={() => void load(listing.parent ?? undefined)}
                    >
                      <ChevronLeft className="size-3.5" />
                      上一级
                    </Button>
                  )}
                  <Button size="sm" variant="ghost" className="h-7 gap-1 px-2 text-[12px]" onClick={() => void load("~")}>
                    <Home className="size-3.5" />
                    家目录
                  </Button>
                  <span className="ml-auto shrink-0 pr-1 text-[11px] text-muted-foreground">
                    {listing.fileCount} 个文件
                  </span>
                </div>

                {listing.dirs.map((dir) => (
                  <button
                    key={dir.path}
                    type="button"
                    onClick={() => void load(dir.path)}
                    className="mt-0.5 flex w-full items-center gap-2 rounded-md px-2 py-2 text-left hover:bg-accent/60"
                  >
                    <FolderOpen className="size-3.5 shrink-0 text-muted-foreground/70" aria-hidden />
                    <span className="min-w-0 flex-1 truncate text-[13px]">{dir.name}</span>
                    <ChevronLeft className="size-3.5 shrink-0 rotate-180 opacity-40" aria-hidden />
                  </button>
                ))}

                {listing.dirs.length === 0 && (
                  <p className="px-2 py-6 text-center text-[12px] text-muted-foreground">这个目录下没有子目录</p>
                )}
              </>
            )}
          </div>

          <div className="shrink-0 border-t px-3 py-2.5">
            <div className="mb-2 truncate font-mono text-[11px] text-muted-foreground">{current}</div>
            <Button className="w-full" disabled={busy || !current} onClick={() => void confirm(current)}>
              {busy ? <Loader2 className="size-4 animate-spin" /> : "用这个目录"}
            </Button>
          </div>
        </>
      )}

  );
}
